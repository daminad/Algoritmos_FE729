Algoritmo P04_SistemaNotas261080017
		
	Definir opcion, ingresado, aprobados Como Entero
	Definir notas, suma, promedio, notamasalta, extra Como Real
	Definir hayNotas Como Logico
	extra = 20
	Dimension notas[5]
	
	hayNotas <- Falso
	
	Repetir
		
		Escribir "===== SISTEMA DE NOTAS FE729 ====="
		Escribir "1. Ingresar notas"
		Escribir "2. Mostrar notas y categorias"
		Escribir "3. Estadisticas"
		Escribir "4. Sumar puntos extra"
		Escribir "5. Salir"
		Escribir "=================================="
		Escribir "Ingrese una opcion:"
		Leer opcion
		
		Segun opcion Hacer
			
			1:
				Escribir "===== INGRESAR NOTAS ====="
				
				Para ingresado <- 1 Hasta 5 Hacer
					
					Escribir "Ingrese la nota del estudiante ", ingresado
					Leer notas[ingresado]
					
					Mientras notas[ingresado] < 0 O notas[ingresado] > 100 Hacer
						Escribir "Nota incorrecta."
						Escribir "Ingrese una nota entre 0 y 100:"
						Leer notas[ingresado]
					FinMientras
					
				FinPara
				
				hayNotas <- Verdadero
				
				Escribir "Notas guardadas correctamente"
				Escribir " "
				Escribir "Presione una tecla para continuar"
				Esperar tecla 
				
			2:
				Si hayNotas = Falso 
					Entonces
					Escribir "Todavia no hay notas."
					Escribir " "
					Escribir "Presione una tecla para continuar"
					Esperar Tecla
						
				Sino
						
					Escribir "===== NOTAS Y CATEGORIAS ====="
						
					Para ingresado <- 1 Hasta 5 Hacer
							
						Si notas[ingresado] >= 90 Entonces
							Escribir "Calificación ", ingresado, ": ", notas[ingresado], " - Excelente"
						Sino
							Si notas[ingresado] >= 80 Entonces
								Escribir "Calificación ", ingresado, ": ", notas[ingresado], " - Muy bueno"
							Sino
								Si notas[ingresado] >= 70 Entonces
									Escribir "Calificación ", ingresado, ": ", notas[ingresado], " - Bueno"
								Sino
									Si notas[ingresado] >= 61 Entonces
										Escribir "Calificación ", ingresado, ": ", notas[ingresado], " - Aprobado"
									Sino
										Escribir "Calificación ", ingresado, ": ", notas[ingresado], " - Reprobado"
									FinSi
								FinSi
							FinSi
						FinSi
						
					FinPara
					Escribir " "
					Escribir "Presione una tecla para continuar"
				Esperar tecla
				FinSi
					
					
			3:
				Si hayNotas = Falso 
					Entonces
					Escribir "Todavia no hay notas."
					Escribir " "
					Escribir "Presione una tecla para continuar"
					Esperar Tecla
						
				Sino
						
					suma <- 0
					aprobados <- 0
					notamasalta <- notas[1]
						
					Para ingresado <- 1 Hasta 5 Hacer
							
						suma <- suma + notas[ingresado]
							
						Si notas[ingresado] > notamasalta Entonces
							notamasalta <- notas[ingresado]
						FinSi
							
						Si notas[ingresado] >= 61 Entonces
							aprobados <- aprobados + 1
						FinSi
							
					FinPara
						
					promedio <- suma / 5
						
					Escribir "===== ESTADISTICAS ====="
					Escribir "Su Promedio es: ", promedio
					Escribir "Su Nota mas alta es: ", notamasalta
					Escribir "Aprobados: ", aprobados
						
				FinSi
				Escribir " "
				Escribir "Presione una tecla para continuar"
				Esperar Tecla
	
					
			4:
				Si hayNotas = Falso 
					Entonces
					Escribir "Todavia no hay notas."
					Escribir " "
					Escribir "Presione una tecla para continuar"
					Esperar Tecla
						
				Sino
					Escribir "===== PUNTOS EXTRA ====="
					Escribir "Tiene 20 puntos extra "
					Escribir " "
					Escribir "Elija una nota para poner puntos extra"
					Escribir "1: " notas[ingresado] 1
					Escribir "2: " notas[ingresado] 2
					Escribir "3: " notas[ingresado] 3
					Escribir "4: " notas[ingresado] 4
					Escribir "5: " notas[ingresado] 5
					leer ingresado
					Esperar Tecla
					
					 ingresado <- ingresado +20
						
						notas[ingresado] <- notas[ingresado] + extra
						
						Si notas[ingresado] > 100 Entonces
							notas[ingresado] <- 100
						FinSi
							
						
					Escribir "Puntos extra agregados."
					Escribir "Las nuevas notas son:"
					
					Para ingresado <- 1 Hasta 5 Hacer
						Escribir "Calificacion ", ingresado, ": ", notas[ingresado]
					FinPara
					Escribir " "
					Escribir "Presione una tecla para continuar"
					Esperar tecla
				FinSi
					
					
			5:
				Escribir "Saliendo del sistema..."
				
			De Otro Modo:
				Escribir "Error, elija una opción válida"
				Escribir " "
				Escribir "Presione una tecla para continuar"
				Esperar tecla
		FinSegun
			
	Hasta Que opcion = 5
		
FinAlgoritmo
